from flask import Flask
import os, threading
from telegram import Update
from telegram.ext import Application, CommandHandler, ContextTypes

app = Flask(__name__)

@app.route('/')
def home():
    return "Bot no ar - Bahia x Remo 20h"

TOKEN = os.getenv("BOT_TOKEN")

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("⚽ Bot ON! Use /palpite")

async def palpite(update: Update, context: ContextTypes.DEFAULT_TYPE):
    texto = """⚽ *BAHIA x REMO - HOJE 20H*

🏆 5º x 19º - Fonte Nova
✅ PRINCIPAL: Bahia vence @1.39
💎 VALOR: Bahia + Over 1.5 @1.85
🔮 Placar: 2-0 Bahia

Bahia 3V seguidas | Remo 8J sem vencer
"""
    await update.message.reply_text(texto, parse_mode='Markdown')

def roda_bot():
    aplic = Application.builder().token(TOKEN).build()
    aplic.add_handler(CommandHandler("start", start))
    aplic.add_handler(CommandHandler("palpite", palpite))
    aplic.run_polling()

threading.Thread(target=roda_bot).start()

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.environ.get("PORT", 10000)))
